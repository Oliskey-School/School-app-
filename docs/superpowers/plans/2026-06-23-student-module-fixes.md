# Student Module Full-Fix Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix all 23 audit findings across the Student Module (7 Critical, 8 High, 6 Medium) and add 3 missing features (grade promotion, withdrawal, auto-ranking), organized into 4 sequential waves.

**Architecture:** All changes are to the Express backend (`backend/src/`) and React frontend (`components/admin/`). Waves 2 and 3 each require one Prisma schema migration. New Wave 4 endpoints follow the existing controller → service → route pattern used throughout the app.

**Tech Stack:** TypeScript, Prisma ORM (PostgreSQL), Express, React, Vitest (integration tests via supertest), `crypto` (Node built-in for password generation).

## Global Constraints

- Every Prisma query touching `Student` or `User` must include `school_id` — never a bare `findUnique` by ID alone without scoping.
- `branchId` resolution: always call `getEffectiveBranchId(req.user, ...)` in controllers; never pass raw header strings to services.
- `school_generated_id` format: `{SCHOOL_CODE}_{BRANCH_CODE}_{ROLE_CODE}_{NUMBER}` — generated by `IdGeneratorService`, never freeform.
- No UI changes outside MED-1 and MED-2 tasks — zero tolerance for layout/style drift.
- Run `npx tsc --noEmit -p backend/tsconfig.json` after every backend task. Zero errors required before commit.
- Run `npm run test:run` after every frontend task.
- Migration command: `npm run db:migrate` (calls `prisma migrate dev`).

export const saveAttendance = async (req: AuthRequest, res: Response) => {
    try {
        const { records } = req.body;
        if (!records || !Array.isArray(records)) {
            return res.status(400).json({ message: 'records array is required' });
        }

        // CRIT-4: teacher may only save attendance for classes they own
        if (req.user.role === 'TEACHER') {
            const teacher = await prisma.teacher.findUnique({
                where: { user_id: req.user.id },
                select: { id: true }
            });
            if (!teacher) return res.status(403).json({ message: 'Teacher profile not found' });

            const uniqueClassIds = [...new Set(
                (records as any[]).map((r: any) => r.class_id).filter(Boolean)
            )];
            if (uniqueClassIds.length > 0) {
                const owned = await prisma.classTeacher.findMany({
                    where: { teacher_id: teacher.id, class_id: { in: uniqueClassIds } },
                    select: { class_id: true }
                });
                const ownedSet = new Set(owned.map(o => o.class_id));
                const unauthorized = uniqueClassIds.filter(id => !ownedSet.has(id));
                if (unauthorized.length > 0) {
                    return res.status(403).json({
                        message: 'Unauthorized: you do not own all submitted classes'
                    });
                }
            }
        }

        const branchId = getEffectiveBranchId(req.user, req.body.branch_id);
        const result = await AttendanceService.saveAttendance(req.user.school_id, branchId, records);
        res.json(result);
    } catch (error: any) {
        res.status(500).json({ message: error.message });
    }
};

static async deleteStudent(schoolId: string, branchId: string | undefined, studentId: string) {
    const student = await prisma.student.findFirst({

static async promoteStudent(
    schoolId: string,
    studentId: string,
    newGrade: number,
    newSection: string,
    branchId?: string,
    session?: string,
    term?: string
): Promise<{ student: any }> {
    const student = await prisma.student.findFirst({
        where: { id: studentId, school_id: schoolId, deleted_at: null }
    });
    if (!student) throw Object.assign(new Error('Student not found'), { status: 404 });

    const effectiveBranchId = branchId || student.branch_id || undefined;

    // 1. Deactivate all current active enrollments
    await prisma.studentEnrollment.updateMany({
        where: { student_id: studentId, school_id: schoolId, status: 'Active' },
        data: { status: 'Inactive' }
    });

    // 2. Find or create the target class
    let targetClass = await prisma.class.findFirst({
        where: {
            school_id: schoolId,
            grade: newGrade,
            section: newSection,
            ...(effectiveBranchId ? { branch_id: effectiveBranchId } : {})
        }
    });

    if (!targetClass) {
        targetClass = await prisma.class.create({
            data: {
                school_id: schoolId,
                branch_id: effectiveBranchId || null,
                grade: newGrade,
                section: newSection,
                name: `Grade ${newGrade} ${newSection}`,
                updated_at: new Date()
            }
        });
    }

    // 3. Create new active enrollment
    await prisma.studentEnrollment.create({
        data: {
            student_id: studentId,
            class_id: targetClass.id,
            school_id: schoolId,
            status: 'Active',
            session: session || '',
            term: term || '',
            updated_at: new Date()
        }
    });

    // 4. Update student grade/section — school_generated_id stays the same
    const updated = await prisma.student.update({
        where: { id: studentId },
        data: { grade: newGrade, section: newSection }
    });

    return { student: updated };
}

```typescript
export const calculateClassRankings = async (req: AuthRequest, res: Response) => {
    try {
        if (!['ADMIN', 'PROPRIETOR', 'TEACHER'].includes(req.user.role)) {
            return res.status(403).json({ message: 'Forbidden' });
        }

        const { classId, term, session } = req.body;
        if (!classId || !term || !session) {
            return res.status(400).json({ message: 'classId, term, and session are required' });
        }

        // 1. Fetch all AcademicPerformance rows for this class/term/session
        const performances = await prisma.academicPerformance.findMany({
            where: {
                class_id: classId,
                term,
                session,
                school_id: req.user.school_id
            },
            select: { id: true, student_id: true, score: true }
        });

        if (performances.length === 0) return res.json([]);

        // 2. Sum scores per student
        const totals = new Map<string, { total: number; ids: string[] }>();
        for (const p of performances) {
            const existing = totals.get(p.student_id) || { total: 0, ids: [] };
            existing.total += (p.score || 0);
            existing.ids.push(p.id);
            totals.set(p.student_id, existing);
        }

        // 3. Sort descending by total score
        const ranked = Array.from(totals.entries())
            .sort((a, b) => b[1].total - a[1].total)
            .map(([studentId, { total, ids }], index) => ({
                studentId,
                totalScore: total,
                position_in_class: index + 1,
                ids
            }));

        // 4. Write position_in_class back to every AcademicPerformance row
        await prisma.$transaction(
            ranked.flatMap(({ position_in_class, ids }) =>
                ids.map(id =>
                    prisma.academicPerformance.update({
                        where: { id },
                        data: { position_in_class }
                    })
                )
            )
        );

        // 5. Return ranked list (omit internal ids array)
        res.json(ranked.map(({ studentId, totalScore, position_in_class }) => ({
            studentId, totalScore, position_in_class
        })));
    } catch (error: any) {
        res.status(500).json({ message: error.message });
    }
};
```
